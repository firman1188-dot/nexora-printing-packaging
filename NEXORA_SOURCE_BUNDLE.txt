# Architecture — NEXORA v5.6

## Arsitektur publik

```text
Beranda
├── Portofolio
├── Material & Finishing
└── Kontak
```

Route dan menu Produk tidak digunakan.

## Integrasi Mobile Control

```text
Mobile Control
      ↓
Portfolio API + Media Storage
      ↓
portfolioApiUrl
      ↓
portfolio.js
      ↓
Filter kategori / material / finishing
      ↓
Grid, preview, spesifikasi, deskripsi
```

## Master Material & Finishing

```text
Master API (opsional)
      ↓
materialFinishingApiUrl
      ↓
material-finishing.js
      ↓
Sample Library + detail + portofolio terkait
```

Fallback lokal:

```text
portfolio-data.js
material-finishing-data.js
```

Kategori, material, dan finishing memakai ID/slug tetap, bukan teks bebas.

# Full-Width Layout Correction Report — v5.1

## Problem found in v5.0

The previous package used `--max: 1220px` for header, content, and footer. On wide monitors—especially when the browser zoom was set to 80%—the website occupied only the center portion of the screen and left large blank margins.

## Correction

- Replaced fixed centered containers with fluid containers and responsive gutters.
- Homepage hero now uses the entire viewport width.
- Inner page heroes use a full-width split composition.
- Page sections keep safe left/right gutters but no restrictive maximum width.
- Large screens receive additional grid columns.
- Mobile uses a separate layout rule set.

## Responsive behavior

### Desktop

- Split hero: text left, product visual right.
- Product visual reaches the right edge of the viewport.
- Eight category icons fit across the bottom homepage rail.
- Four-column category index on wide screens.

### Mobile

- Compact fixed header with hamburger menu.
- Two-column hero composition.
- Product image cropped intentionally for visual impact.
- Category icons arranged in two ordered rows.
- Inner page heroes stack text and image vertically.

## Files changed

- `src/assets/css/styles.css`
- `package.json`
- Documentation and QA references

async function buildDirectory(sourceDir, outputDir) {
  await mkdir(outputDir, { recursive: true });
  const entries = await readdir(sourceDir, { withFileTypes: true });
  for (const entry of entries) {
    const sourcePath = path.join(sourceDir, entry.name);
    const outputPath = path.join(outputDir, entry.name);
    if (entry.isDirectory()) {
      await buildDirectory(sourcePath, outputPath);
      continue;
    }
    let content = await readFile(sourcePath, 'utf8');
    content = content
      .replaceAll('{{HOME_HEADER}}', homeHeader)
      .replaceAll('{{HEADER}}', header)
      .replaceAll('{{FOOTER}}', footer)
      .replaceAll('{{CONSULTATION}}', consultation);
    await writeFile(outputPath, content);
  }
}

await buildDirectory(path.join(src, 'pages'), dist);

const sitemapPages = [

import { createServer } from 'node:http';
import { readFile, stat } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const dist = path.join(root, 'dist');
const port = Number(process.env.PORT || 3000);
spawnSync(process.execPath, [path.join(root, 'scripts/build.mjs')], { stdio: 'inherit' });

const types = { '.html':'text/html; charset=utf-8', '.css':'text/css; charset=utf-8', '.js':'text/javascript; charset=utf-8', '.svg':'image/svg+xml', '.webp':'image/webp', '.png':'image/png', '.jpg':'image/jpeg', '.xml':'application/xml', '.txt':'text/plain; charset=utf-8', '.webmanifest':'application/manifest+json' };

function safePath(urlPath) {
  const clean = decodeURIComponent(urlPath.split('?')[0]);
  if (clean.includes('..')) return null;
  if (clean === '/') return path.join(dist, 'index.html');
  const direct = path.join(dist, clean);
  return direct;
}

async function resolveFile(urlPath) {
  let candidate = safePath(urlPath);
  if (!candidate) return null;
  try { if ((await stat(candidate)).isFile()) return candidate; } catch {}
  if (!path.extname(candidate)) {
    try { if ((await stat(`${candidate}.html`)).isFile()) return `${candidate}.html`; } catch {}
    try { if ((await stat(path.join(candidate, 'index.html'))).isFile()) return path.join(candidate, 'index.html'); } catch {}
  }
  return null;
}

createServer(async (req, res) => {
  const file = await resolveFile(req.url || '/');
  const target = file || path.join(dist, '404.html');
  try {
    const data = await readFile(target);
    res.writeHead(file ? 200 : 404, { 'Content-Type': types[path.extname(target)] || 'application/octet-stream', 'Cache-Control': 'no-store' });
    res.end(data);
  } catch (error) {
    res.writeHead(500, { 'Content-Type': 'text/plain' });
    res.end(String(error));
  }
}).listen(port, '0.0.0.0', () => {
  console.log(`NEXORA running at http://localhost:${port}`);
});

import { readFile, stat } from 'node:fs/promises';
import path from 'node:path';
import vm from 'node:vm';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');

const errors = [];

const context = { window: {} };
vm.createContext(context);
vm.runInContext(await readFile(path.join(src, 'assets/js/material-finishing-data.js'), 'utf8'), context);
vm.runInContext(await readFile(path.join(src, 'assets/js/portfolio-data.js'), 'utf8'), context);

const master = context.window.NEXORA_MATERIAL_FINISHING_DATA;
const portfolio = context.window.NEXORA_PORTFOLIO_DATA;
if (!master) errors.push('NEXORA_MATERIAL_FINISHING_DATA tidak tersedia.');

const unique = (values, label) => {
  const seen = new Set();
  for (const value of values) {
    if (!value) errors.push(`${label}: nilai kosong.`);
    if (seen.has(value)) errors.push(`${label}: duplikat ${value}.`);
    seen.add(value);
  }
};

const materials = master?.materials || [];
const finishings = master?.finishings || [];
unique(materials.map((item) => item.material_id), 'material_id');
unique(materials.map((item) => item.material_slug), 'material_slug');
unique(finishings.map((item) => item.finishing_id), 'finishing_id');
unique(finishings.map((item) => item.finishing_slug), 'finishing_slug');

const portfolioMaterialIds = new Set((portfolio?.materials || []).map((item) => item.material_id));
const portfolioFinishingIds = new Set((portfolio?.finishings || []).map((item) => item.finishing_id));

import { readFile, readdir, stat } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const dist = path.join(root, 'dist');
let errors = [];

async function walk(dir) {
  const out = [];
  for (const entry of await readdir(dir, { withFileTypes: true })) {
    const p = path.join(dir, entry.name);
    if (entry.isDirectory()) out.push(...await walk(p)); else out.push(p);
  }
  return out;
}

const files = await walk(dist);
const htmlFiles = files.filter((f) => f.endsWith('.html'));
const assetRefs = new Set();

for (const file of htmlFiles) {
  const html = await readFile(file, 'utf8');
  if (html.includes('{{HEADER}}') || html.includes('{{FOOTER}}')) errors.push(`${file}: unreplaced template token`);
  if (!html.includes('<meta name="viewport"')) errors.push(`${file}: missing viewport`);
  for (const match of html.matchAll(/(?:src|href)="(\/assets\/[^"#?]+)"/g)) assetRefs.add(match[1]);
  if (/three\.js|webgl|\.glb/i.test(html)) errors.push(`${file}: forbidden legacy 3D reference`);
}

for (const ref of assetRefs) {
  const target = path.join(dist, ref);
  try { if (!(await stat(target)).isFile()) errors.push(`Missing asset: ${ref}`); } catch { errors.push(`Missing asset: ${ref}`); }
}

const css = await readFile(path.join(dist, 'assets/css/styles.css'), 'utf8');
if (!css.includes('height: 100dvh')) errors.push('Homepage viewport lock missing');
if (/threeProduct|glbPilot|WebGL/i.test(css)) errors.push('Legacy 3D CSS remains');
const js = await readFile(path.join(dist, 'assets/js/main.js'), 'utf8');
if (/THREE|GLTFLoader|WebGLRenderer/.test(js)) errors.push('Legacy 3D JavaScript remains');

if (errors.length) {
  console.error(`Verification failed (${errors.length}):`);
  errors.forEach((e) => console.error(`- ${e}`));
  process.exit(1);
}
console.log(`Verification PASS: ${htmlFiles.length} HTML files, ${assetRefs.size} referenced assets.`);

(() => {
  const config = window.NEXORA_CONFIG || {};
  const body = document.body;

  // Header states
  const header = document.querySelector('[data-site-header]');
  const megaWrap = document.querySelector('[data-mega-wrap]');
  const megaTrigger = document.querySelector('[data-mega-trigger]');
  const mobileToggle = document.querySelector('[data-mobile-toggle]');
  const mobileMenu = document.querySelector('[data-mobile-menu]');

  const closeMega = () => {
    megaWrap?.classList.remove('open');
    megaTrigger?.setAttribute('aria-expanded', 'false');
  };

  megaTrigger?.addEventListener('click', (event) => {
    event.stopPropagation();
    const isOpen = megaWrap.classList.toggle('open');
    megaTrigger.setAttribute('aria-expanded', String(isOpen));
  });

  document.addEventListener('click', (event) => {
    if (megaWrap && !megaWrap.contains(event.target)) closeMega();
  });

  mobileToggle?.addEventListener('click', () => {
    const isOpen = mobileMenu.classList.toggle('open');
    mobileToggle.classList.toggle('open', isOpen);
    mobileToggle.setAttribute('aria-expanded', String(isOpen));
    body.classList.toggle('menu-open', isOpen);
  });

  window.addEventListener('scroll', () => {
    header?.classList.toggle('scrolled', window.scrollY > 16);
  }, { passive: true });

  // Active navigation
  const path = window.location.pathname;
  if (path === '/' || path.endsWith('/index.html')) document.querySelector('[data-nav="home"]')?.classList.add('active');
  if (path.includes('portofolio')) document.querySelector('[data-nav="portfolio"]')?.classList.add('active');
  if (path.includes('kontak')) document.querySelector('[data-nav="contact"]')?.classList.add('active');
  if (path.includes('material-finishing')) document.querySelector('[data-nav="material"]')?.classList.add('active');

  // Business data
  document.querySelectorAll('[data-current-year]').forEach((node) => node.textContent = new Date().getFullYear());

    node.href = `https://wa.me/${config.whatsapp || ''}`;
  });
  document.querySelectorAll('[data-business-email]').forEach((node) => {
    node.textContent = config.email || 'info@nexora.id';
    node.href = `mailto:${config.email || 'info@nexora.id'}`;
  });
  document.querySelectorAll('[data-business-address]').forEach((node) => node.textContent = config.address || 'Bandung, Jawa Barat');
  document.querySelectorAll('[data-map-link]').forEach((node) => node.href = config.mapUrl || '#');

  // Consultation wizard
  const modal = document.querySelector('[data-consultation-modal]');
  const form = document.querySelector('[data-consultation-form]');
  const steps = [...document.querySelectorAll('[data-step]')];
  const progress = [...document.querySelectorAll('.consult-progress span')];
  const nextButton = document.querySelector('[data-consult-next]');
  const backButton = document.querySelector('[data-consult-back]');
  const submitButton = document.querySelector('[data-consult-submit]');
  let activeStep = 1;

  const updateWizard = () => {
    steps.forEach((step) => step.classList.toggle('active', Number(step.dataset.step) === activeStep));
    progress.forEach((bar, index) => bar.classList.toggle('active', index < activeStep));
    if (backButton) backButton.disabled = activeStep === 1;
    if (nextButton) nextButton.hidden = activeStep === steps.length;
    if (submitButton) submitButton.hidden = activeStep !== steps.length;
  };

  const openModal = () => {
    modal?.classList.add('open');
    modal?.setAttribute('aria-hidden', 'false');
    body.classList.add('modal-open');
    setTimeout(() => modal?.querySelector('input,button')?.focus(), 60);
  };

  const closeModal = () => {
    modal?.classList.remove('open');
    modal?.setAttribute('aria-hidden', 'true');
    body.classList.remove('modal-open');
  };

  document.querySelectorAll('[data-open-consultation]').forEach((button) => button.addEventListener('click', openModal));
  document.querySelectorAll('[data-close-consultation]').forEach((button) => button.addEventListener('click', closeModal));

  nextButton?.addEventListener('click', () => {
    const current = steps.find((step) => Number(step.dataset.step) === activeStep);

    const valid = required.every((input) => input.type === 'radio'

      : input.checkValidity());
    if (!valid) {
      current.querySelector('[required]')?.reportValidity();
      return;
    }
    activeStep = Math.min(steps.length, activeStep + 1);
    updateWizard();
  });

    activeStep = Math.max(1, activeStep - 1);
    updateWizard();

        gallery.querySelectorAll('[data-gallery-src]').forEach((item) => item.classList.remove('active'));

    lightbox.classList.remove('open');
    body.classList.remove('modal-open');
  }));

        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return normalizePortfolio(await response.json());
      } catch (error) {
        console.warn('Portfolio API gagal pada Material & Finishing, menggunakan data lokal.', error);
      }
    }
    return normalizePortfolio(window.NEXORA_PORTFOLIO_DATA || {});
  };

  const loadData = async () => {
    if (config.materialFinishingApiUrl) {
      try {
        const response = await fetch(config.materialFinishingApiUrl, { headers: { Accept: 'application/json' } });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const payload = await response.json();
        return payload?.data || payload;
      } catch (error) {
        console.warn('Master Material & Finishing API gagal, menggunakan data lokal.', error);
      }
    }
    return window.NEXORA_MATERIAL_FINISHING_DATA || { materials: [], finishings: [], material_groups: [], finishing_groups: [] };
  };

  const activeItems = () => itemsForMode()
    .filter((item) => String(item.status || 'ACTIVE').toUpperCase() === 'ACTIVE')
    .filter((item) => state.group === 'all' || item[groupKey()] === state.group)
    .sort((a, b) => Number(a.display_order || 9999) - Number(b.display_order || 9999));

  const currentItem = () => itemsForMode().find((item) => item[idKey()] === state.selectedId);

  const groupName = (groupId) => groupsForMode().find((group) => group.group_id === groupId)?.group_name || '';

  const updateUrl = () => {
    const item = currentItem();
    const url = new URL(window.location.href);
    url.searchParams.set('type', state.mode);
    if (state.group !== 'all') url.searchParams.set('group', state.group); else url.searchParams.delete('group');
    if (item) url.searchParams.set('id', item[slugKey()]); else url.searchParams.delete('id');
    try { history.replaceState({}, '', `${url.pathname}${url.search}`); } catch (error) { if (error?.name !== 'SecurityError') console.warn(error); }
  };

  const renderModes = () => {
    elements.modeButtons.forEach((button) => {
      const active = button.dataset.mfMode === state.mode;
      button.classList.toggle('active', active);
      button.setAttribute('aria-selected', String(active));
    });
  };

  const renderFilters = () => {
    elements.filters.innerHTML = groupsForMode().map((group) => `
      <button type="button" class="mf-filter${group.group_id === state.group ? ' active' : ''}" data-mf-group="${escapeHtml(group.group_id)}">
        <span class="mf-filter-icon">${icon(group.icon)}</span>
        <span>${escapeHtml(group.group_name)}</span>
      </button>`).join('');

    elements.filters.querySelectorAll('[data-mf-group]').forEach((button) => {
      button.addEventListener('click', () => {
        state.group = button.dataset.mfGroup;
        state.page = 1;
        state.selectedId = null;
        render();
      });
    });
  };

  const renderGrid = (items) => {
    const totalPages = Math.max(1, Math.ceil(items.length / state.pageSize));
    state.page = Math.min(state.page, totalPages);
    const start = (state.page - 1) * state.pageSize;
    const pageItems = items.slice(start, start + state.pageSize);

    if (!pageItems.length) {
      elements.grid.innerHTML = `<div class="mf-empty"><span>${icon('grid')}</span><h2>Belum ada sampel</h2><p>Belum ada data pada kelompok yang dipilih.</p></div>`;
      elements.resultText.textContent = '0 sampel';
      return;
    }

    elements.grid.innerHTML = pageItems.map((item) => {
      const id = item[idKey()];
      const active = id === state.selectedId;
      return `
        <article class="mf-card${active ? ' active' : ''}" data-mf-id="${escapeHtml(id)}">
          <button type="button" aria-label="Lihat detail ${escapeHtml(item[nameKey()])}">
            <span class="mf-card-image"><img src="${escapeHtml(item.preview_image)}" alt="${escapeHtml(item[nameKey()])}" loading="lazy"></span>
            <span class="mf-card-copy"><b>${escapeHtml(item[nameKey()])}</b><small>${escapeHtml(groupName(item[groupKey()]))}</small></span>
            <span class="mf-card-check">${icon('check')}</span>
          </button>
        </article>`;
    }).join('');

    elements.grid.querySelectorAll('[data-mf-id] button').forEach((button) => {
      button.addEventListener('click', () => selectItem(button.closest('[data-mf-id]').dataset.mfId, true));
    });

    elements.resultText.textContent = `Menampilkan ${start + 1}–${Math.min(start + pageItems.length, items.length)} dari ${items.length} ${state.mode === 'material' ? 'material' : 'finishing'}`;
  };

  const renderPagination = (items) => {
    const totalPages = Math.max(1, Math.ceil(items.length / state.pageSize));
    if (totalPages <= 1) {
      elements.pagination.innerHTML = '';
      return;
    }
    elements.pagination.innerHTML = `
      <button type="button" class="page-arrow" data-page="${Math.max(1, state.page - 1)}" ${state.page === 1 ? 'disabled' : ''}>${icon('chevron-left')}</button>
      ${Array.from({ length: totalPages }, (_, index) => index + 1).map((page) => `<button type="button" class="page-number${page === state.page ? ' active' : ''}" data-page="${page}">${page}</button>`).join('')}
      <button type="button" class="page-arrow" data-page="${Math.min(totalPages, state.page + 1)}" ${state.page === totalPages ? 'disabled' : ''}>${icon('chevron-right')}</button>`;
    elements.pagination.querySelectorAll('[data-page]').forEach((button) => button.addEventListener('click', () => {
      state.page = Number(button.dataset.page);
      state.selectedId = null;
      render();
    }));
  };

  const relatedPortfolios = (item) => state.portfolios.filter((portfolio) => state.mode === 'material'
    ? (portfolio.material_ids || []).includes(item.material_id)
    : (portfolio.finishing_ids || []).includes(item.finishing_id));

  const portfolioUrl = (item, portfolio) => {
    const filter = state.mode === 'material' ? `material=${encodeURIComponent(item.material_slug)}` : `finishing=${encodeURIComponent(item.finishing_slug)}`;
    const selected = portfolio ? `&item=${encodeURIComponent(portfolio.slug)}` : '';
    return `/portofolio.html?${filter}${selected}`;
  };

  const materialSpecs = (item) => [
    ['Jenis', item.type || '-'], ['Gramatur', item.grammage || '-'], ['Permukaan', item.surface || '-'], ['Warna', item.color || '-'], ['Ketebalan', item.thickness || '-']
  ];

  const renderDetail = () => {
    const item = currentItem();
    if (!item) {
      elements.detail.innerHTML = `<div class="mf-detail-placeholder"><span>${icon('material')}</span><h2>Pilih sampel</h2><p>Klik salah satu card untuk melihat preview dan informasi lengkap.</p></div>`;
      elements.detail.classList.remove('open');
      return;
    }

    const related = relatedPortfolios(item);
    const isMaterial = state.mode === 'material';
    const specs = isMaterial ? materialSpecs(item) : [['Efek Utama', item.effect || '-'], ['Kelompok', groupName(item.finishing_group)]];
    const characteristics = item.characteristics || [];
    const uses = item.recommended_uses || [];
    const combinations = item.recommended_combinations || [];
    const actionTitle = isMaterial ? 'Butuh Bahan Ini?' : 'Butuh Finishing Ini?';

    elements.detail.innerHTML = `
      <button type="button" class="mf-detail-close" data-mf-detail-close aria-label="Tutup detail">${icon('close')}</button>
      <div class="mf-detail-preview"><img src="${escapeHtml(item.preview_image)}" alt="${escapeHtml(item[nameKey()])}"></div>
      <div class="mf-detail-heading">
        <div><h2>${escapeHtml(item[nameKey()])}</h2><p>${escapeHtml(groupName(item[groupKey()]))}</p></div>
        <button type="button" class="mf-share" data-mf-share>${icon('share')}<span>Bagikan</span></button>
      </div>

      <div class="mf-detail-scroll">
        <div class="mf-info-columns">
          <section>
            <h3>${isMaterial ? 'Spesifikasi' : 'Informasi Finishing'}</h3>
            <dl>${specs.map(([label, value]) => `<div><dt>${escapeHtml(label)}</dt><dd>${escapeHtml(value)}</dd></div>`).join('')}</dl>
          </section>
          <section>
            <h3>Karakteristik</h3>
            <ul class="mf-check-list">${characteristics.map((value) => `<li>${icon('check')}<span>${escapeHtml(value)}</span></li>`).join('')}</ul>
          </section>
        </div>

        <section class="mf-uses">
          <h3>Cocok Untuk</h3>
          <div>${uses.map((value, index) => `<span>${icon(['box','book','label','calendar','bag','promo'][index % 6])}<b>${escapeHtml(value)}</b></span>`).join('')}</div>
        </section>

        ${!isMaterial && combinations.length ? `<section class="mf-combinations"><h3>Kombinasi yang Disarankan</h3><div>${combinations.map((value) => `<span>${escapeHtml(value)}</span>`).join('')}</div></section>` : ''}

        <section class="mf-notes"><h3>Catatan</h3><p>${escapeHtml(item.notes || item.short_description || '-')}</p></section>

        <section class="mf-related">
          <div class="mf-related-heading"><h3>Hasil Produksi Terkait</h3><a href="${portfolioUrl(item)}">Lihat Semua ${icon('arrow')}</a></div>
          ${related.length ? `<div class="mf-related-grid">${related.slice(0, 4).map((portfolio) => `<a href="${portfolioUrl(item, portfolio)}" title="${escapeHtml(portfolio.title)}"><img src="${escapeHtml(portfolio.cover_image)}" alt="${escapeHtml(portfolio.title)}"><span>${escapeHtml(portfolio.title)}</span></a>`).join('')}</div>` : '<p class="mf-related-empty">Belum ada portofolio aktif yang menggunakan pilihan ini.</p>'}
        </section>
      </div>

      <div class="mf-detail-actions">
        <button type="button" class="btn btn-orange" data-mf-consult>${icon('whatsapp')}<span><b>${actionTitle}</b><small>Konsultasi via WhatsApp</small></span></button>
        <a class="btn btn-outline" href="${portfolioUrl(item)}"><span><b>Lihat Portofolio</b><small>Produk menggunakan pilihan ini</small></span>${icon('arrow')}</a>
      </div>`;

    elements.detail.classList.add('open');
    elements.detail.querySelector('[data-mf-detail-close]')?.addEventListener('click', closeDetail);
    elements.detail.querySelector('[data-mf-share]')?.addEventListener('click', shareItem);
    elements.detail.querySelector('[data-mf-consult]')?.addEventListener('click', () => openConsultation(item));
  };

  const selectItem = (id, openMobile = false) => {
    state.selectedId = id;
    renderGrid(activeItems());
    renderDetail();
    updateUrl();
    if (openMobile && window.matchMedia('(max-width: 899px)').matches) {
      document.body.classList.add('mf-detail-open');
      setTimeout(() => elements.detail.querySelector('[data-mf-detail-close]')?.focus(), 40);
    }
  };

  const closeDetail = () => {
    document.body.classList.remove('mf-detail-open');
    if (window.matchMedia('(max-width: 899px)').matches) return;
    state.selectedId = null;
    renderGrid(activeItems());
    renderDetail();
    updateUrl();
  };

  const shareItem = async () => {
    const item = currentItem();
    if (!item) return;
    updateUrl();
    const data = { title: `${item[nameKey()]} — NEXORA`, text: item.short_description || item.effect || '', url: window.location.href };
    try {
      if (navigator.share) await navigator.share(data);
      else await navigator.clipboard.writeText(window.location.href);
    } catch (error) { if (error?.name !== 'AbortError') console.warn(error); }
  };

  const openConsultation = (item) => {
    const kind = state.mode === 'material' ? 'material' : 'finishing';
    const message = [
      `Halo NEXORA, saya ingin berkonsultasi mengenai ${kind}:`, '',
      `${state.mode === 'material' ? 'Material' : 'Finishing'}: ${item[nameKey()]}`,
      `Referensi: ${window.location.href}`, '',
      'Mohon bantu rekomendasi spesifikasi dan penerapannya.'
    ].join('\n');
    window.open(`https://wa.me/${config.whatsapp || ''}?text=${encodeURIComponent(message)}`, '_blank', 'noopener');
  };

  const render = () => {
    renderModes();
    renderFilters();
    const items = activeItems();
    if (!state.selectedId || !items.some((item) => item[idKey()] === state.selectedId)) state.selectedId = items[0]?.[idKey()] || null;
    renderGrid(items);
    renderPagination(items);
    renderDetail();
    updateUrl();
  };

  const applyUrl = () => {
    const params = new URLSearchParams(window.location.search);
    state.mode = params.get('type') === 'finishing' ? 'finishing' : 'material';
    const groups = groupsForMode();
    const requestedGroup = params.get('group');
    state.group = groups.some((group) => group.group_id === requestedGroup) ? requestedGroup : 'all';
    const requestedSlug = params.get('id');
    const selected = itemsForMode().find((item) => item[slugKey()] === requestedSlug);
    state.selectedId = selected?.[idKey()] || null;
  };

  const init = async () => {
    elements.status.textContent = 'Memuat sampel…';
    state.pageSize = resolvePageSize();
    state.data = await loadData();
    state.portfolios = await loadPortfolio();
    applyUrl();
    elements.status.textContent = '';

    elements.modeButtons.forEach((button) => button.addEventListener('click', () => {
      state.mode = button.dataset.mfMode;
      state.group = 'all';
      state.page = 1;
      state.selectedId = null;
      document.body.classList.remove('mf-detail-open');
      render();
    }));

    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape' && document.body.classList.contains('mf-detail-open')) closeDetail();
    });

    let resizeTimer;
    window.addEventListener('resize', () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(() => {
        const next = resolvePageSize();
        if (next === state.pageSize) return;
        state.pageSize = next;
        state.page = 1;
        render();
      }, 120);
    });

    render();
  };

  init().catch((error) => {
    console.error(error);
    elements.status.innerHTML = '<strong>Material & Finishing gagal dimuat.</strong>';
  });
})();

(() => {
  const root = document.querySelector('[data-portfolio-app]');
  if (!root) return;

  const config = window.NEXORA_CONFIG || {};
  const state = {
    data: null,
    category: 'all',
    material: null,
    finishing: null,
    page: 1,
    pageSize: 12,
    selectedId: null,
    selectedImageIndex: 0,
    detailOpen: false
  };


  const resolvePageSize = () => {
    if (window.matchMedia('(max-width: 899px)').matches) return 4;
    if (window.matchMedia('(max-width: 1280px)').matches) return 6;
    return 12;
  };

  const elements = {
    filters: root.querySelector('[data-portfolio-filters]'),
    context: root.querySelector('[data-portfolio-filter-context]'),
    grid: root.querySelector('[data-portfolio-grid]'),
    resultText: root.querySelector('[data-portfolio-result-text]'),
    pagination: root.querySelector('[data-portfolio-pagination]'),
    detail: root.querySelector('[data-portfolio-detail]'),
    status: root.querySelector('[data-portfolio-status]')
  };

  const escapeHtml = (value = '') => String(value)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');

  const formatQuantity = (value, unit) => {
    if (value === null || value === undefined || value === '') return '-';
    return `${new Intl.NumberFormat('id-ID').format(Number(value))} ${unit || ''}`.trim();
  };

  const icon = (name) => window.NEXORA_ICONS?.[name] || '';

  const normalizeData = (payload) => {
    const source = payload?.data || payload || {};
    const categories = source.categories || source.portfolio_categories || [];
    const materials = source.materials || source.master_materials || [];
    const finishings = source.finishings || source.master_finishings || [];
    const items = source.portfolio_items || source.items || source.portfolios || [];

    return {
      categories,
      materials,
      finishings,
      portfolio_items: items
        .filter((item) => String(item.status || 'ACTIVE').toUpperCase() === 'ACTIVE')
        .sort((a, b) => Number(a.display_order || 9999) - Number(b.display_order || 9999))
    };
  };

  const loadData = async () => {
    if (config.portfolioApiUrl) {
      try {
        const response = await fetch(config.portfolioApiUrl, { headers: { Accept: 'application/json' } });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return normalizeData(await response.json());
      } catch (error) {
        console.warn('Portfolio API gagal, menggunakan data lokal.', error);
      }
    }
    return normalizeData(window.NEXORA_PORTFOLIO_DATA || {});
  };

  const categoryById = (id) => state.data.categories.find((item) => item.category_id === id);
  const categoryBySlug = (slug) => state.data.categories.find((item) => item.category_slug === slug);
  const materialById = (id) => state.data.materials.find((item) => item.material_id === id);
  const materialBySlug = (slug) => state.data.materials.find((item) => item.material_slug === slug);
  const finishingById = (id) => state.data.finishings.find((item) => item.finishing_id === id);
  const finishingBySlug = (slug) => state.data.finishings.find((item) => item.finishing_slug === slug);

  const getFilteredItems = () => state.data.portfolio_items.filter((item) => {
    const categoryMatch = state.category === 'all' || item.category_id === state.category;
    const materialMatch = !state.material || (item.material_ids || []).includes(state.material);
    const finishingMatch = !state.finishing || (item.finishing_ids || []).includes(state.finishing);
    return categoryMatch && materialMatch && finishingMatch;
  });

  const updateUrl = () => {
    const url = new URL(window.location.href);
    const category = state.category === 'all' ? null : categoryById(state.category)?.category_slug;
    const material = state.material ? materialById(state.material)?.material_slug : null;
    const finishing = state.finishing ? finishingById(state.finishing)?.finishing_slug : null;
    const item = state.selectedId ? state.data.portfolio_items.find((entry) => entry.portfolio_id === state.selectedId)?.slug : null;

    [['category', category], ['material', material], ['finishing', finishing], ['item', item]].forEach(([key, value]) => {
      if (value) url.searchParams.set(key, value); else url.searchParams.delete(key);
    });
    try {
      history.replaceState({}, '', `${url.pathname}${url.search}${url.hash}`);
    } catch (error) {
      // Opaque/file preview origins may block history mutation; UI remains functional.
      if (error?.name !== 'SecurityError') console.warn('Gagal memperbarui URL portofolio.', error);
    }
  };

  const renderFilters = () => {
    const filters = [
      { category_id: 'all', category_name: 'Semua', icon: 'grid' },
      ...state.data.categories
    ];

    elements.filters.innerHTML = filters.map((category) => `
      <button type="button" class="portfolio-filter${state.category === category.category_id ? ' active' : ''}" data-category-id="${escapeHtml(category.category_id)}">
        <span class="portfolio-filter-icon">${icon(category.icon)}</span>
        <span>${escapeHtml(category.category_name)}</span>
      </button>
    `).join('');

    elements.filters.querySelectorAll('[data-category-id]').forEach((button) => {
      button.addEventListener('click', () => {
        state.category = button.dataset.categoryId;
        state.page = 1;
        state.selectedId = null;
        state.selectedImageIndex = 0;
        render();
      });
    });
  };

  const renderFilterContext = () => {
    const chips = [];
    if (state.material) {
      const item = materialById(state.material);
      chips.push(`<button type="button" data-clear-filter="material">Material: <b>${escapeHtml(item?.material_name || state.material)}</b> ×</button>`);
    }
    if (state.finishing) {
      const item = finishingById(state.finishing);
      chips.push(`<button type="button" data-clear-filter="finishing">Finishing: <b>${escapeHtml(item?.finishing_name || state.finishing)}</b> ×</button>`);
    }
    elements.context.innerHTML = chips.length ? `<span>Filter aktif:</span>${chips.join('')}` : '';
    elements.context.hidden = chips.length === 0;
    elements.context.querySelectorAll('[data-clear-filter]').forEach((button) => {
      button.addEventListener('click', () => {
        state[button.dataset.clearFilter] = null;
        state.page = 1;
        render();
      });
    });
  };

  const renderGrid = (items) => {
    const totalPages = Math.max(1, Math.ceil(items.length / state.pageSize));
    state.page = Math.min(state.page, totalPages);
    const start = (state.page - 1) * state.pageSize;
    const pageItems = items.slice(start, start + state.pageSize);

    if (!pageItems.length) {
      elements.grid.innerHTML = `
        <div class="portfolio-empty">
          <span class="portfolio-empty-icon">${icon('grid')}</span>
          <h2>Belum ada hasil produksi</h2>
          <p>Belum ada portofolio yang memenuhi seluruh filter yang dipilih.</p>
          <button type="button" class="btn btn-orange" data-open-consultation>Konsultasi Sekarang</button>
        </div>`;
      elements.resultText.textContent = '0 portofolio';
      elements.grid.querySelector('[data-open-consultation]')?.addEventListener('click', () => document.querySelector('.header-actions [data-open-consultation]')?.click());
      return;
    }

    elements.grid.innerHTML = pageItems.map((item) => {
      const category = categoryById(item.category_id);
      const active = item.portfolio_id === state.selectedId;
      return `
        <article class="portfolio-result-card${active ? ' active' : ''}" data-portfolio-id="${escapeHtml(item.portfolio_id)}">
          <button type="button" aria-label="Lihat detail ${escapeHtml(item.title)}">
            <span class="portfolio-card-image"><img src="${escapeHtml(item.cover_image)}" alt="${escapeHtml(item.image_alt_text || item.title)}" loading="lazy"></span>
            <span class="portfolio-card-copy">
              <b>${escapeHtml(item.title)}</b>
              <small>${escapeHtml(category?.category_name || '')}</small>
            </span>
            <span class="portfolio-card-check" aria-hidden="true">${icon('check')}</span>
          </button>
        </article>`;
    }).join('');

    elements.grid.querySelectorAll('[data-portfolio-id] button').forEach((button) => {
      button.addEventListener('click', () => {
        const card = button.closest('[data-portfolio-id]');
        selectItem(card.dataset.portfolioId, true);
      });
    });

    const first = start + 1;
    const last = Math.min(start + pageItems.length, items.length);
    elements.resultText.textContent = `Menampilkan ${first}–${last} dari ${items.length} portofolio`;
  };

  const renderPagination = (items) => {
    const totalPages = Math.max(1, Math.ceil(items.length / state.pageSize));
    if (totalPages <= 1) {
      elements.pagination.innerHTML = '';
      return;
    }

    const pages = Array.from({ length: totalPages }, (_, index) => index + 1);
    elements.pagination.innerHTML = `
      <button type="button" class="page-arrow" data-page="${Math.max(1, state.page - 1)}" ${state.page === 1 ? 'disabled' : ''} aria-label="Halaman sebelumnya">${icon('chevron-left')}</button>
      ${pages.map((page) => `<button type="button" class="page-number${page === state.page ? ' active' : ''}" data-page="${page}">${page}</button>`).join('')}
      <button type="button" class="page-arrow" data-page="${Math.min(totalPages, state.page + 1)}" ${state.page === totalPages ? 'disabled' : ''} aria-label="Halaman berikutnya">${icon('chevron-right')}</button>`;

    elements.pagination.querySelectorAll('[data-page]').forEach((button) => {
      button.addEventListener('click', () => {
        state.page = Number(button.dataset.page);
        state.selectedId = null;
        render();
      });
    });
  };

  const detailImage = (item) => (item.gallery_images?.length ? item.gallery_images : [item.cover_image]);

  const renderDetail = () => {
    const item = state.data.portfolio_items.find((entry) => entry.portfolio_id === state.selectedId);
    if (!item) {
      elements.detail.innerHTML = `
        <div class="portfolio-detail-placeholder">
          <span>${icon('grid')}</span>
          <h2>Pilih hasil produksi</h2>
          <p>Klik salah satu foto untuk melihat preview besar dan informasi teknisnya.</p>
        </div>`;
      elements.detail.classList.remove('open');
      return;
    }

    const category = categoryById(item.category_id);
    const materials = (item.material_ids || []).map((id) => materialById(id)?.material_name).filter(Boolean);
    const finishings = (item.finishing_ids || []).map((id) => finishingById(id)?.finishing_name).filter(Boolean);
    const images = detailImage(item);
    state.selectedImageIndex = Math.min(state.selectedImageIndex, images.length - 1);
    const currentImage = images[state.selectedImageIndex] || item.cover_image;
    const additional = [...(item.additional_specifications || [])].sort((a, b) => Number(a.display_order || 0) - Number(b.display_order || 0));

    elements.detail.innerHTML = `
      <button type="button" class="portfolio-detail-close" data-detail-close aria-label="Tutup detail">${icon('close')}</button>
      <div class="portfolio-preview">
        <img src="${escapeHtml(currentImage)}" alt="${escapeHtml(item.image_alt_text || item.title)}">
        ${images.length > 1 ? `
          <button type="button" class="preview-arrow prev" data-image-step="-1" aria-label="Foto sebelumnya">${icon('chevron-left')}</button>
          <button type="button" class="preview-arrow next" data-image-step="1" aria-label="Foto berikutnya">${icon('chevron-right')}</button>
          <div class="preview-dots">${images.map((_, index) => `<button type="button" class="${index === state.selectedImageIndex ? 'active' : ''}" data-image-index="${index}" aria-label="Foto ${index + 1}"></button>`).join('')}</div>` : ''}
      </div>

      <div class="portfolio-detail-heading">
        <div>
          <h2>${escapeHtml(item.title)}</h2>
          <p>${escapeHtml(category?.category_name || '')}</p>
        </div>
        <button type="button" class="portfolio-share" data-share-portfolio>${icon('share')}<span>Bagikan</span></button>
      </div>

      <section class="portfolio-spec-card" aria-labelledby="portfolio-spec-title">

            <div><dt>${icon('size')}<span>Ukuran</span></dt><dd>${escapeHtml(item.size?.display || item.size || '-')}</dd></div>
            <div><dt>${icon('print')}<span>Cetak</span></dt><dd>${escapeHtml(item.printing_method || '-')}</dd></div>
            <div><dt>${icon('finish')}<span>Finishing</span></dt><dd>${escapeHtml(finishings.join(', ') || '-')}</dd></div>
            <div><dt>${icon('quantity')}<span>Jumlah</span></dt><dd>${escapeHtml(formatQuantity(item.production_quantity, item.production_unit))}</dd></div>
          </dl>
          <div class="portfolio-extra-specs">
            <h4>Spesifikasi Tambahan</h4>
            ${additional.length ? `<ul>${additional.map((spec) => `<li><span>${escapeHtml(spec.name)}</span><b>${escapeHtml(spec.value)}</b></li>`).join('')}</ul>` : '<p>Tidak ada spesifikasi tambahan.</p>'}
          </div>
        </div>

      <section class="portfolio-description">
        <h3>Deskripsi Produk</h3>
        <p>${escapeHtml(item.description || '-')}</p>
      </section>

      <div class="portfolio-detail-actions">
        <button type="button" class="btn btn-orange" data-similar-product>${icon('whatsapp')}Buat Produk Serupa</button>

      button.addEventListener('click', () => {
        state.selectedImageIndex = (state.selectedImageIndex + Number(button.dataset.imageStep) + images.length) % images.length;
        renderDetail();
      });
    });
    elements.detail.querySelectorAll('[data-image-index]').forEach((button) => {
      button.addEventListener('click', () => {
        state.selectedImageIndex = Number(button.dataset.imageIndex);
        renderDetail();
      });
    });
    elements.detail.querySelector('[data-share-portfolio]')?.addEventListener('click', shareSelected);
    elements.detail.querySelector('[data-similar-product]')?.addEventListener('click', () => openSimilarProduct(item));
    elements.detail.querySelector('[data-open-consultation]')?.addEventListener('click', () => {
      document.querySelector('.header-actions [data-open-consultation]')?.click();
    });
  };

  const selectItem = (id, openMobile = false) => {
    state.selectedId = id;
    state.selectedImageIndex = 0;
    state.detailOpen = true;
    renderGrid(getFilteredItems());

    updateUrl();

      document.body.classList.add('portfolio-detail-open');

      elements.detail.classList.remove('open');
      return;

    state.selectedImageIndex = 0;
    renderGrid(getFilteredItems());
    renderDetail();
    updateUrl();

  const shareSelected = async () => {
    const item = state.data.portfolio_items.find((entry) => entry.portfolio_id === state.selectedId);
    if (!item) return;
    updateUrl();
    const shareData = { title: `${item.title} — NEXORA`, text: item.description, url: window.location.href };
    try {
      if (navigator.share) await navigator.share(shareData);
      else {
        await navigator.clipboard.writeText(window.location.href);
        const button = elements.detail.querySelector('[data-share-portfolio] span');
        if (button) {
          const original = button.textContent;
          button.textContent = 'Tautan disalin';
          setTimeout(() => button.textContent = original, 1600);
        }
      }
    } catch (error) {
      if (error?.name !== 'AbortError') console.warn('Gagal membagikan portofolio.', error);
    }
  };

  const openSimilarProduct = (item) => {
    const category = categoryById(item.category_id)?.category_name || '-';
    const message = [
      'Halo NEXORA, saya tertarik membuat produk serupa dengan portofolio berikut:',
      '',
      `Produk: ${item.title}`,
      `Kategori: ${category}`,
      `Referensi: ${window.location.href}`,
      '',
      'Mohon bantu konsultasi spesifikasi dan penawarannya.'
    ].join('\n');
    window.open(`https://wa.me/${config.whatsapp || ''}?text=${encodeURIComponent(message)}`, '_blank', 'noopener');
  };

  const render = () => {
    renderFilters();
    renderFilterContext();
    const filtered = getFilteredItems();

    if (!state.selectedId || !filtered.some((item) => item.portfolio_id === state.selectedId)) {
      state.selectedId = filtered[0]?.portfolio_id || null;
      state.selectedImageIndex = 0;
    }

    renderGrid(filtered);
    renderPagination(filtered);
    renderDetail();
    updateUrl();
  };

  const applyUrlFilters = () => {
    const params = new URLSearchParams(window.location.search);
    const category = categoryBySlug(params.get('category'));
    const material = materialBySlug(params.get('material'));
    const finishing = finishingBySlug(params.get('finishing'));
    const selected = state.data.portfolio_items.find((item) => item.slug === params.get('item'));

    state.category = category?.category_id || 'all';
    state.material = material?.material_id || null;
    state.finishing = finishing?.finishing_id || null;
    state.selectedId = selected?.portfolio_id || null;
  };

  const init = async () => {
    elements.status.textContent = 'Memuat portofolio…';
    state.pageSize = resolvePageSize();
    state.data = await loadData();
    applyUrlFilters();
    elements.status.textContent = '';
    render();

    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape' && document.body.classList.contains('portfolio-detail-open')) closeDetail();
    });

    let resizeTimer;
    window.addEventListener('resize', () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(() => {
        const nextPageSize = resolvePageSize();
        if (nextPageSize === state.pageSize) return;
        state.pageSize = nextPageSize;
        state.page = 1;
        render();
      }, 120);
    });
  };

  init().catch((error) => {
    console.error(error);
    elements.status.innerHTML = '<strong>Portofolio gagal dimuat.</strong><br>Periksa sumber data atau koneksi API.';
  });
})();
